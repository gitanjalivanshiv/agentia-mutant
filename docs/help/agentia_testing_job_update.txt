Update Copado Robotic Testing job/test metadata

USAGE
  $ agentia testing job update ID -p <value> [--json] [--git-branch <value>]
    [--git-domain-credential <value>] [--git-password <value> |
    --git-password-file <value> | --git-password-stdin] [--git-tag <value>]
    [--git-url <value>] [--git-use-caller-credentials] [--git-username <value>]
    [--case-duration-change-detection] [--description <value>]
    [--hide-video-params] [--name <value>] [--no-case-duration-change-detection]
    [--parallel-mode disabled|enabled] [--robot <value>] [--screen-depth
    <value>] [--screen-height <value>] [--screen-width <value>]
    [--show-video-params] [--suite-type
    bitbar|browserstack|browserstackdesktop|default|lambdatest|lambdatestdesktop
    |mobile|paceconnectmobile|privatemobile|rpa|selenese|zap] [--timeout
    <value>]

ARGUMENTS
  ID  CRT job/test ID.

FLAGS
  -p, --project=<value>                    (required) CRT project ID.
      --case-duration-change-detection     Enable case duration change detection
                                           analytics.
      --description=<value>                Job/test description.
      --git-branch=<value>                 Git branch to use for a Git-backed
                                           CRT job. Mutually exclusive with
                                           --git-tag.
      --git-domain-credential=<value>      CRT domain credential ID to use for
                                           Git repository access.
      --git-password=<value>               Git password/token for repository
                                           access. Prefer
                                           --git-domain-credential when
                                           available.
      --git-password-file=<value>          Read a bounded Git password/token
                                           from a regular file.
      --git-password-stdin                 Read a Git password/token from stdin.
      --git-tag=<value>                    Git tag to use for a Git-backed CRT
                                           job. Mutually exclusive with
                                           --git-branch.
      --git-url=<value>                    Git repository URL for a Git-backed
                                           CRT job.
      --git-use-caller-credentials         Use caller credentials for Git
                                           repository access instead of explicit
                                           Git username/password.
      --git-username=<value>               Git username for repository access.
                                           Prefer --git-domain-credential when
                                           available.
      --hide-video-params                  Hide video parameters.
      --name=<value>                       Job/test name.
      --no-case-duration-change-detection  Disable case duration change
                                           detection analytics.
      --parallel-mode=<option>             Parallel execution mode.
                                           <options: disabled|enabled>
      --robot=<value>                      CRT robot ID that should execute this
                                           job/test.
      --screen-depth=<value>               Screen color depth. Must be provided
                                           with --screen-width and
                                           --screen-height.
      --screen-height=<value>              Screen height. Must be provided with
                                           --screen-width and --screen-depth.
      --screen-width=<value>               Screen width. Must be provided with
                                           --screen-height and --screen-depth.
      --show-video-params                  Show video parameters.
      --suite-type=<option>                CRT suite/job type.
                                           <options: bitbar|browserstack|browser
                                           stackdesktop|default|lambdatest|lambd
                                           atestdesktop|mobile|paceconnectmobile
                                           |privatemobile|rpa|selenese|zap>
      --timeout=<value>                    Build timeout, for example 10s, 10m,
                                           10h, or 10d.

GLOBAL FLAGS
  --json  Format output as json.

DESCRIPTION
  Update Copado Robotic Testing job/test metadata

EXAMPLES
  $ agentia testing job update <job-id> --project <project-id> --name "Smoke"

  $ agentia testing job update <job-id> --project <project-id> --git-url https://github.com/acme/crt-tests.git --git-branch main

  $ agentia testing job update <job-id> --project <project-id> --git-branch feature/login-smoke

  $ agentia testing job update <job-id> --project <project-id> --timeout 10m --parallel-mode enabled --json
